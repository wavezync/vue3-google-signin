## Quick start

<Tabs syncKey="pkg">
<TabItem label="npm">

```sh
npm install vue3-google-signin
```

</TabItem>
<TabItem label="pnpm">

```sh
pnpm add vue3-google-signin
```

</TabItem>
<TabItem label="yarn">

```sh
yarn add vue3-google-signin
```

</TabItem>
</Tabs>

```ts title="main.ts"
import { createApp } from "vue";
import GoogleSignInPlugin from "vue3-google-signin";
import App from "./App.vue";

createApp(App)
  .use(GoogleSignInPlugin, {
    clientId: "YOUR_CLIENT_ID.apps.googleusercontent.com",
  })
  .mount("#app");
```

```vue title="LoginButton.vue"
<script setup lang="ts">
import {
  GoogleSignInButton,
  decodeCredential,
  type CredentialResponse,
} from "vue3-google-signin";

function onSuccess({ credential }: CredentialResponse) {
  const user = decodeCredential(credential);
  console.log(`Signed in as ${user.email}`);
}
</script>

<template>
  <GoogleSignInButton @success="onSuccess" />
</template>
```

Continue with the [guide](/guide/) for One Tap, custom buttons and Nuxt.
